# tools/secret_patterns.txt: the ONE secret pattern list (MISSION V11 §5.3).
# Used by .githooks/pre-commit (via tools/redact.py), tools/redact.py, tools/safe_add.sh
# and tools/mission_check.py (#6, #15). Patterns describe SHAPES only; never a real value.
#
# Format: <kind> <name> <pattern>    (pattern = rest of the line)
#   name    : case-insensitive glob on a file's BASE name (path-only check; such files are never opened)
#   content : Python regex, applied per line (secrets)
#   pii     : Python regex, applied per line (personal data: phone, PAN, email)
#   exempt  : glob on a base name that name-rules never apply to: code files are scanned by
#             CONTENT instead (e.g. src/bot/login/secrets.py, a future token_manager.py)
# The pre-commit hook and safe_add.sh refuse name + content hits; the redactor removes content + pii hits;
# the PII scan (mission_check #10, with tools/scan_allowlist.toml) arrives in T02b.
# 40-hex git SHAs and 64-hex digests are never matched by these shapes (hash context stays intact).

exempt code_py *.py
exempt code_go *.go

name dotenv .env*
name pem *.pem
name key *.key
name p12 *.p12
name pfx *.pfx
name ssh_rsa id_rsa*
name ssh_ed25519 id_ed25519*
name secret_ext *.secret
name secrets_file secrets.*
name credentials_json credentials*.json
name token_json *token*.json
name token_txt *token*.txt
name access_token_file access_token*

content pem_private -----BEGIN ([A-Z0-9]+ )*PRIVATE KEY-----
content jwt_like \beyJ[A-Za-z0-9_-]{20,}
content auth_header (?i)\bauthorization\b["']?\s*[:=]\s*["']?(bearer|basic|token)\s+[A-Za-z0-9._~+/=-]{8,}
content bearer_value (?i)\bbearer\s+[A-Za-z0-9._~+/=-]{20,}
content kv_secret_quoted (?i)\b(access_token|refresh_token|auth_token|api_key|apikey|api_secret|client_secret|secret_key|private_key|password|passwd)\b["']?\s*[:=]\s*["'][^"'\s]{8,}["']
content kv_secret_env (?i)^\s*(export\s+)?[A-Z0-9_]*(TOKEN|SECRET|PASSWORD|API_KEY|APIKEY)[A-Z0-9_]*\s*=\s*[^\s$"'{(<\[]{12,}\s*$
content url_token (?i)[?&](access_token|api_key|apikey|token|client_secret)=[A-Za-z0-9._~+/-]{12,}

pii phone_in (?<![0-9A-Za-z])(\+91[-\s]?)?[6-9][0-9]{9}(?![0-9A-Za-z])
pii pan_in (?<![A-Za-z0-9])[A-Z]{3}[ABCFGHLJPT][A-Z][0-9]{4}[A-Z](?![A-Za-z0-9])
pii email (?<![A-Za-z0-9._%+-])[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,}
